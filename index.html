function drawFloor(chars, floorIdx, cx, ctop, rx, ryT, CR, STICK, GAP) { if (chars.length === 0) return; const n = chars.length; const slotH = CR * 2 + 30; const sl = STICK + floorIdx * slotH; const scale = 1 - floorIdx * 0.08; const frx = rx * 0.88 * scale; const fry = ryT * 0.88 * scale; const angleOffset = floorIdx % 2 === 0 ? -Math.PI/2 : -Math.PI/2 + Math.PI/n; // 위치 미리 계산 const positions = []; for (let i = 0; i < n; i++) { const a = angleOffset + (2 * Math.PI / n) * i; const px = cx + Math.cos(a) * frx; const py = ctop + Math.sin(a) * fry; const icy = py - sl - CR; positions.push({ px, py, icy }); } // ✅ 1단계: 막대 + 불꽃 먼저 전부 그리기 positions.forEach(({ px, py, icy }) => { /* 초 막대 */ ctx.save(); const sw = 7; const gg = ctx.createLinearGradient(px-sw/2, py, px+sw/2, py); gg.addColorStop(0,'#E8B090'); gg.addColorStop(.4,'#C8956C'); gg.addColorStop(1,'#A07050'); ctx.strokeStyle=gg; ctx.lineWidth=sw; ctx.lineCap='round'; ctx.beginPath(); ctx.moveTo(px,py); ctx.lineTo(px,py-sl); ctx.stroke(); ctx.restore(); /* 불꽃 */ ctx.save(); ctx.globalAlpha=.92; ctx.beginPath(); ctx.ellipse(px,py-sl-11,4,10,0,0,Math.PI*2); ctx.fillStyle='#FFD700'; ctx.fill(); ctx.beginPath(); ctx.ellipse(px,py-sl-10,2.5,6,0,0,Math.PI*2); ctx.fillStyle='#FF8C00'; ctx.fill(); ctx.globalAlpha=1; ctx.restore(); }); // ✅ 2단계: 사진 + 이름 전부 그리기 (막대 위에) positions.forEach(({ px, py, icy }, i) => { /* 흰 테두리 */ ctx.save(); ctx.beginPath(); ctx.arc(px,icy,CR+4,0,Math.PI*2); ctx.fillStyle='#fff'; ctx.shadowColor='rgba(242,92,138,.4)'; ctx.shadowBlur=16; ctx.fill(); ctx.restore(); ctx.save(); ctx.beginPath(); ctx.arc(px,icy,CR+4,0,Math.PI*2); ctx.strokeStyle='#FF9BBB'; ctx.lineWidth=3; ctx.stroke(); ctx.restore(); /* 사진 */ const globalIdx = allChars.filter(c=>c.imgData).indexOf(chars[i]); const img = getCachedImg(chars[i].imgData); if (img) { ctx.save(); ctx.beginPath(); ctx.arc(px,icy,CR,0,Math.PI*2); ctx.clip(); ctx.drawImage(img,px-CR,icy-CR,CR*2,CR*2); ctx.restore(); } else { ctx.save(); ctx.beginPath(); ctx.arc(px,icy,CR,0,Math.PI*2); ctx.fillStyle='rgba(255,155,187,.3)'; ctx.fill(); ctx.restore(); } /* 이름 */ const nm = (chars[i].name||'').slice(0,8) + ((chars[i].name?.length||0)>8?'…':''); if (nm) { ctx.font='bold 13px "Gmarket Sans",sans-serif'; const tw = ctx.measureText(nm).width; const lx=px-tw/2-7, ly=icy+CR+6; roundRectFill(ctx, lx, ly, tw+14, 21, 8, 'rgba(255,107,157,.18)'); ctx.fillStyle='#D94A78'; ctx.textAlign='center'; ctx.fillText(nm, px, ly+15); } /* 닉네임 */ if (chars[i].nick) { ctx.font='11px "Gmarket Sans",sans-serif'; ctx.fillStyle='rgba(155,122,172,.85)'; ctx.textAlign='center'; ctx.fillText(chars[i].nick, px, icy-CR-9); } hitAreas.push({x:px, y:icy, r:CR+4, idx: globalIdx}); }); }
